func runtimeOriginal62Root(t *testing.T) string {
 t.Helper();root:=os.Getenv("OMABIT_APP_RUNTIME_UPGRADE_ROOT")
 if !filepath.IsAbs(root)||filepath.Clean(root)!=root||!strings.HasPrefix(root,"/tmp/omabit-app-runtime-upgrade-") {t.Fatal("explicit disposable source fixture required")}
 return root
}
