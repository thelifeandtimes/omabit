  function resourceCanConfigure(target){
    if(!target || busy || !resourceReady)return false
    var c=connections.filter(function(c){return c.name===target.connection && c.runner_id===target.runner})[0]
    return Model.writable(c,now) && c.snapshot.instances.some(function(i){return i.id===target.instance && !i.project_id && i.config_revision===target.revision})
  }
  function resourceConfigure(target,port,loom,memory,cpus){
    if(!resourceCanConfigure(target))return false
    try {Model.configure(target,port,loom,memory,cpus)}catch(e){lastError=e.message;return false}
    return resourceSubmit("configure",{target:target,config:{http_port:Number(port),ames_port:target.ames,loom:Number(loom),memory_gib:Number(memory),cpu_cores:Number(cpus)}})
  }
  function resourceOperate(connection,instance,kind){
    if(["start","stop"].indexOf(kind)<0 || instance.project_id || !Model.canOperate(connection,instance,kind,now))return false
    return resourceSubmit(kind,{target:Model.edit(connection,instance)})
